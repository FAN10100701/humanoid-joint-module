📜 核心源码(完整可跑,约 20 行)

// 价值噪声 → 5 层分形叠加(fbm):高频细节 + 低频轮廓
float fbm(vec2 p){ float v=0., a=.5;
  for(int i=0;i<5;i++){ v += a*noise(p); p*=2.03; a*=.5; }
  return v; }
void main(){
  vec2 uv = gl_FragCoord.xy/uRes;
  float col = 0.;
  for(int i=0;i<3;i++){                    // 三列雨柱(对应三相电流)
    float x = uv.x*(1.+float(i)*.4) + float(i)*1.7;
    float n = fbm(vec2(x*8., uT*uB*.3 - uv.y*2. + float(i)*9.));
    col += pow(n, 3.) * (1.-uv.y) * smoothstep(0.,.15,uv.y);
  }
  col *= uA;
  vec3 c = vec3(.02,.03,.05)
         + vec3(.45,.6,.9)*col             // 冷色主体
         + vec3(.2,.8,.9)*pow(col,3.)*.6;  // 高光溅射
  gl_FragColor = vec4(c, 1.);
}

要点:①噪声输入的 y 分量减去 uT → 图案向下平流;②pow(n,3) 拉开明暗对比,只有噪声峰值存活成"雨丝";③三层不同缩放/偏移的雨柱叠加出纵深。拖动下方滑杆感受 uA(强度)与 uB(流速)两个 uniform 的作用。
