// 圆环 SDF:到"半径 1.4 的圆环、管径 0.06"的有向距离
float sdTorus(vec3 p, vec2 t){
  vec2 q = vec2(length(p.xz)-t.x, p.y);
  return length(q)-t.y;
}
float map(vec3 p, float dens){
  float d = 1e9;
  for(int i=0;i<4;i++){
    vec3 q = p;
    q.z += uT*uB*2. + float(i)*7.2/dens; // 沿 z 排开并飞向相机
    q.z = mod(q.z, 7.2) - 3.6;           // 无限循环走廊
    d = min(d, sdTorus(q, vec2(1.4,.06)));
  }
  return d;
}
void main(){
  vec3 ro = vec3(0.,0.,-3.), rd = normalize(vec3(uv,.85));
  float t = 0., glow = 0.;
  for(int i=0;i<64;i++){
    float d = map(ro+rd*t, dens);
    glow += .02/(1. + d*d*40.);   // 越近越亮 → 体积辉光
    t += max(d*.8, .012);
    if(t>22.) break;
  }
  vec3 c = vec3(.1,.5,1.)*glow + vec3(.55,.2,1.)*glow*glow*2.2;
  gl_FragColor = vec4(c*vignette, 1.);
}

要点:①mod(q.z,7.2) 把空间折叠成无限循环;②辉光不是"打光"而是累计 1/d²——距离场天然给出"多接近物体";③蓝紫双色按 glow/glow² 混合出霓虹质感。整个世界只有一段距离函数。
