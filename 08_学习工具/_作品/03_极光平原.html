📜 核心源码(完整可跑,约 14 行)

// 极光带:水平 fbm 曲线 + 垂直高斯衰减
float band(vec2 uv, float t){
  float ridge = fbm(vec2(uv.x*2.2, t*.35))*.6 + uv.x*.15;
  float d = uv.y - (.55 + ridge*.35);   // 距极光脊线的距离
  return exp(-abs(d)*6.);               // 高斯衰减 → 柔和光带
}
vec3 ramp(float v){                     // 亮度 → 色(青→紫→品红)
  vec3 c = vec3(.0,.35,.3);
  c = mix(c, vec3(.35,.1,.55), smoothstep(.2,.7,v));
  c = mix(c, vec3(.9,.25,.6),  smoothstep(.7,1.,v));
  return c;
}
void main(){
  float a = band(uv,uT)*uA;
  float mountain = smoothstep(.34,.28, uv.y + fbm(uv*3.)*.06);
  vec3 c = ramp(min(a,1.))*a + vec3(.012,.02,.045);
  c = mix(c, vec3(.006,.01,.018), mountain);  // 雪山剪影
  c += vec3(.9)*stars(uv);                    // 顶层星点
  gl_FragColor = vec4(c,1.);
}

要点:①exp(-|d|·k) 高斯光带比线性衰减自然得多;②ramp() 三段 mix 是"色域映射"最小实现;③雪山就是一行 smoothstep 阈值——shader 里"画山"只需要一个不等式;④星点 = 网格哈希 + 距离衰减 + 呼吸闪烁。
